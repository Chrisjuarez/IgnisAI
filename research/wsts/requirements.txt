# Do NOT pip install these into `ignis-ml` (the training env) or into conda
# `base`. earthaccess pulls s3fs -> aiobotocore -> a newer botocore, which
# conflicts with boto3 and breaks S3 access:
#
#     boto3 1.40.20 requires botocore<1.41.0, but earthaccess wants >=1.43.3
#
# That is a real conflict, not a warning to click past. Keeping this track in
# its own environment is the whole reason it is a "parallel track".
#
#   conda create -n wsts python=3.11 -y
#   conda activate wsts
#   pip install -r research/wsts/requirements.txt
#
# Verify with:  python -c "import earthaccess, rasterio, pyproj, torch; print('ok')"
